"""The FP8 policy must not silently convert teacher or auxiliary objectives."""

from unittest import mock

import pytest
import torch
from torch import nn

from specforge.training.torchtitan.quantization import (
    convert_draft_float8,
    float8_candidate_names,
)


def _draft():
    draft = nn.Module()
    draft.fc = nn.Linear(256, 128, bias=False)
    layer = nn.Module()
    layer.self_attn = nn.Module()
    layer.self_attn.q_proj = nn.Linear(128, 128, bias=False)
    layer.self_attn.k_proj = nn.Linear(128, 64, bias=False)
    layer.self_attn.v_proj = nn.Linear(128, 128, bias=False)
    layer.self_attn.v_proj.requires_grad_(False)
    layer.self_attn.o_proj = nn.Linear(128, 130, bias=False)
    layer.mlp = nn.Module()
    layer.mlp.gate_proj = nn.Linear(128, 256, bias=False)
    layer.mlp.up_proj = nn.Linear(128, 256, bias=False)
    layer.mlp.down_proj = nn.Linear(256, 128, bias=False)
    layer.mlp_conv = nn.Module()
    layer.mlp_conv.kernel_projection = nn.Linear(128, 256, bias=False)
    draft.layers = nn.ModuleList([layer])
    draft.candidate_selector = nn.Module()
    draft.candidate_selector.hidden_projection = nn.Linear(128, 128, bias=False)
    draft.markov_head = nn.Module()
    draft.markov_head.markov_w2 = nn.Linear(128, 256, bias=False)
    draft.lm_head = nn.Linear(128, 256, bias=False)
    return draft


def test_candidates_exclude_frozen_small_unaligned_and_auxiliary_modules():
    assert float8_candidate_names(_draft()) == (
        "fc",
        "layers.0.self_attn.q_proj",
        "layers.0.mlp.gate_proj",
        "layers.0.mlp.up_proj",
        "layers.0.mlp.down_proj",
    )


def test_no_phantom_fp8_when_no_projection_is_eligible():
    with pytest.raises(ValueError, match="no eligible"):
        convert_draft_float8(_draft(), min_dimension=1024)


def test_rejects_invalid_recipe_and_threshold():
    with pytest.raises(ValueError, match="recipe"):
        convert_draft_float8(_draft(), recipe_name="mxfp8")
    with pytest.raises(ValueError, match="at least 16"):
        float8_candidate_names(_draft(), min_dimension=1)


def test_rejects_unsupported_hardware_before_importing_torchao():
    with mock.patch("torch.cuda.is_available", return_value=False):
        with pytest.raises(RuntimeError, match="SM89"):
            convert_draft_float8(_draft())


def test_rejects_unvalidated_torchao_version_before_conversion():
    with (
        mock.patch("torch.cuda.is_available", return_value=True),
        mock.patch("torch.cuda.get_device_capability", return_value=(9, 0)),
        mock.patch(
            "specforge.training.torchtitan.quantization.version", return_value="0.19.0"
        ),
    ):
        with pytest.raises(RuntimeError, match="requires torchao==0.18.0"):
            convert_draft_float8(_draft())


@pytest.mark.skipif(not torch.cuda.is_available(), reason="requires an FP8 GPU")
def test_conversion_preserves_parameters_names_and_auxiliary_precision():
    pytest.importorskip("torchao")
    from torchao.float8.float8_linear import Float8Linear

    draft = _draft().eval()
    before = dict(draft.named_parameters())
    names = convert_draft_float8(draft)
    after = dict(draft.named_parameters())
    assert before.keys() == after.keys()
    assert all(after[name] is parameter for name, parameter in before.items())
    for name in names:
        module = draft.get_submodule(name)
        assert isinstance(module, Float8Linear)
        assert not module.training
        assert module.config.pad_inner_dim
        assert not module.config.enable_fsdp_float8_all_gather
    assert type(draft.lm_head) is nn.Linear
    assert type(draft.markov_head.markov_w2) is nn.Linear
    assert type(draft.candidate_selector.hidden_projection) is nn.Linear
    assert type(draft.layers[0].mlp_conv.kernel_projection) is nn.Linear
    restored = _draft()
    restored.load_state_dict(draft.state_dict(), strict=True)
    for name, value in restored.state_dict().items():
        torch.testing.assert_close(value, draft.state_dict()[name], rtol=0, atol=0)
